ti_decimation = {

	enable = yes
	toggle = no
	army_only = yes
	
	map = yes
	
	soundeffect ="event:/SFX/UI/Unit/sfx_ui_unit_reorganization"


	allow = {
		is_army = yes
		in_combat = no
		is_moving = no
		is_exiled = no
		has_commander = yes
		unit_owner = {
			exists = yes
			or = {
				invention = land_morale_recovery_inv_5
				tag = ROM
			}
		}
		every_sub_unit = {
			subunit_strength_percentage > 0.1
		}
		trigger_if = {
			limit = {
				exists = commander
			}
			commander = {
				loyalty > 33
			}
		}
	}
	
	
	start_effect = {
		legion = {
			remove_distinction = distinction_dishonor_cowardly
		}
		every_sub_unit = {
			add_subunit_strength = -0.1
			add_subunit_morale = -1
		}
		commander = {
			add_trait = cruel
			remove_trait = kindly
			add_trait = disciplined
		}
		add_unit_modifier =  { 
			name = ti_decimation_modifier
			duration = 180
		}
	}
	
	
	ai_will_do = {
		modifier = {
			trigger = {
				legion = {
					has_distinction = distinction_dishonor_cowardly
				}
				invention = land_morale_recovery_inv_5
			}

			add = {
				value = 250
				divide = unit_size_rank
			}
		}
	}
}

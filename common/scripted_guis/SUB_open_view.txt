show_player_subjects_button = {
	scope = country
	ai_is_valid = {
		always = no
	}
	effect = {
		every_subject = {
			root = {
				add_to_variable_list = {
					name = player_subjects
					target = prev
				}
			}
		}
	}
}

remove_player_subjects_variables_button = {
	ai_is_valid = { always = no }
	scope = country
	effect = {
        every_in_list = {
            variable = player_subjects
            root = {
	            remove_list_variable = {
	                name = player_subjects
	                target = prev
	            }
	        }
        }
	}
}

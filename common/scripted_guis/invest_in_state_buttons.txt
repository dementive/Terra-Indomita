invest_in_state_button_administration = {
    scope = state

	confirm_title = "invest_admin_t"
	confirm_text = "invest_admin_desc"

	saved_scopes = {
		player
	}

	is_shown = { # Can be omitted as always true

	}

	ai_is_valid = {
		this = scope:player.capital_scope.state
		scope:player = {
			OR = {
				AND = {
					can_pay_price = price_state_investment_administration
					political_influence >= 200
				}
				has_variable = free_investments
			}
		}
	}

	ai_chance = {
		base = 1
	}

  	is_valid = {
  		trigger_if = {
  			limit = {
  				NOT = { exists = root }
  			}
  			exists = root
  		}
		trigger_if = {
			limit = {
				scope:player = {
					OR = {
						NOT = {
							has_variable = free_investments
						}
						AND = {
							has_variable = free_investments
							var:free_investments = 0
						}
					}
					NOT = {
						can_pay_price = price_state_investment_administration
					}
				}
			}
			hidden:scope:player = {
				can_pay_price = price_state_investment_administration
			}
		}
		trigger_if = {
			limit = {
				has_state_modifier = state_modification_in_progress
			}
			custom_tooltip = {
				text = "already_improving_infrastructure"
				NOT = {
					has_state_modifier = state_modification_in_progress
				}
			}
		}
		add_scaling_gold_cost_trigger = yes
    }

	effect = {
		scope:player = {
			if = {
				limit = {
					has_variable = free_investments
					var:free_investments > 0
				}
				custom_tooltip = "reduce_investments_by_1"
				change_variable = {
					name = free_investments
					add = -1
				}
			}
			else = {
				pay_price = price_state_investment_administration
				root = { add_scaling_gold_cost_effect = yes }
			}
		}
		custom_tooltip = investments_ongoing_1_year
		hidden_effect = {
			add_state_modifier = {
				name = state_modification_in_progress
				duration = 182
			}
			save_scope_as = state_improvement_target
			trigger_event = {
				id = state_improvement.19
				days = 182
			}
			add_state_modification_value_effect = yes
		}
		show_as_tooltip = {
			add_state_modifier = {
				name = state_improvement_admin
				duration = -1
				mode = add
			}
		}
	}
}

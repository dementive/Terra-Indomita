	republican_mediterranean_1 = { #citizen navy
		modifier = {

			trireme_discipline = 0.05
			global_ship_start_experience = 10
		}
	}
	republican_mediterranean_2 = {#shipbuilding initiative
		modifier = {
			ship_cost = -0.05
			global_ship_recruit_speed = 0.2
			war_exhaustion = 0.02
		}
	}
	republican_mediterranean_3 = {#shrewd business tactics
		allow  = {
			num_of_ports >= 1
			or = {
				invention = trireme_discipline_inv_1
			}
		}
		on_enact = {
			remove_variable = root_out_pirates
			if = {
				limit = {
					any_owned_province = {
						NOT = { has_province_modifier = pirate_haven_modifier }
					}
				}
				custom_tooltip = pirate_will_be_placed_tt
				hidden_effect = {
					random_owned_province = {
						limit = {
							is_port = yes
						}
						weight = {
							modifier = {
								factor = 1000
								controller = root
							}
						}
						add_permanent_province_modifier = {
							name = pirate_haven_modifier
							duration = -1
						}
					}
				}
			}
		}
		modifier = {
			global_export_commerce_modifier = -0.1
			global_import_commerce_modifier = 0.2
			diplomatic_reputation = -1
			mercenary_naval_maintenance_cost = -0.1
		}
	}
	republican_mediterranean_4 = { #anti priary
		allow = {
			num_of_ports >= 1
			invention = trireme_discipline_inv_1
		}
		on_enact = {
			custom_tooltip = enables_root_out_pirates_tt
			set_variable = {
				name = root_out_pirates
				days = -1
			}
		}
		modifier = {
			anti_piracy_cb = yes
			global_citizen_happyness = 0.04
			naval_morale = 0.1
		}
	}
	republican_mediterranean_5 = { #thalassacracy

		}
		modifier = {
